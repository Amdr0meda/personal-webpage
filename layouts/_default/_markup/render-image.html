{{- /* Imágenes del contenido Markdown. - Busca la imagen en el bundle del post (o en assets/). - Genera WebP en varios anchos (srcset) para que cada pantalla descargue solo lo que necesita. - Ancho y alto explícitos para evitar saltos de maquetación; carga diferida. */ -}} {{- $dest := .Destination -}} {{- $u := urls.Parse $dest -}} {{- $img := "" -}} {{- if not $u.IsAbs -}} {{- $path := strings.TrimPrefix "./" $u.Path -}} {{- $img = or (.Page.Resources.Get $path) (resources.Get $path) -}} {{- end -}} {{- if and $img (in (slice "png" "jpeg" "jpg" "webp") $img.MediaType.SubType) -}} {{- $max := cond (gt $img.Width 1440) 1440 $img.Width -}} {{- $widths := slice -}} {{- range slice 480 800 1200 -}} {{- if lt . $max }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- $widths = $widths | append $max -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $widths -}} {{- $r := $img.Resize (printf "%dx webp q78 drawing" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- if or (not $fallback) (le . 800) }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- $tag := printf `` $fallback.RelPermalink (delimit $srcset ", ") $img.Width $img.Height (.Text | plainify) -}} {{- if and .IsBlock .Title -}} {{- else -}} {{ $tag | safeHTML }} {{- end -}} {{- else -}} {{- end -}}
